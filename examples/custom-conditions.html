📝 自定义规则配置源码

const customObserver = createBreakpointObserver({
  widthBreakpoints: {
    'Mobile (<600px)': '< 600px',
    'Tablet Landscape (AND: 768-1024px)': { and: ['>= 768px', '<= 1024px'] },
    'Extreme Screens (OR: <360 or >=1920)': { or: ['< 360px', '>= 1920px'] },
    'Rem Unit (>= 50rem)': '>= 50rem',
    'Not Exact Desktop (!= 1200px)': '!= 1200px'
  }
})
